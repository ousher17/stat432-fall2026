lambda_grid <- 10^seq(-4, 0, length.out = 41)

fit_lasso <- function(X, y, seed = 1) {

  set.seed(seed)

  n <- nrow(X)

  folds <- sample(rep(1:10, length.out = n))
  cv_errors <- matrix(
    NA,
    nrow = 10,
    ncol = length(lambda_grid)
  )
  for (fold in 1:10) {
    train_index <- which(folds != fold)
    valid_index <- which(folds == fold)
    X_train <- X[train_index, , drop = FALSE]
    X_valid <- X[valid_index, , drop = FALSE]
    y_train <- y[train_index]
    y_valid <- y[valid_index]
    x_mean <- colMeans(X_train)
    y_mean <- mean(y_train)
    x_scale <- sqrt(
      colMeans(
        sweep(X_train, 2, x_mean, "-")^2
      )
    )

    X_train_std <- sweep(
      X_train, 2, x_mean, "-"
    )

    X_train_std <- sweep(
      X_train_std, 2, x_scale, "/"
    )

    y_train_centered <- y_train - y_mean

    X_valid_std <- sweep(
      X_valid, 2, x_mean, "-"
    )

    X_valid_std <- sweep(
      X_valid_std, 2, x_scale, "/"
    )

    for (j in seq_along(lambda_grid)) {
      fit <- glmnet(
        X_train_std,
        y_train_centered,
        alpha = 1,
        lambda = lambda_grid[j],
        standardize = FALSE,
        intercept = FALSE
      )
      pred <- y_mean +
        as.numeric(
          predict(
            fit,
            newx = X_valid_std,
            s = lambda_grid[j]
          )
        )

      cv_errors[fold, j] <- mean(
        (y_valid - pred)^2
      )
    }
  }
  mean_cv_error <- colMeans(cv_errors)
  best_lambda <- lambda_grid[
    which.min(mean_cv_error)
  ]
  x_mean <- colMeans(X)
  y_mean <- mean(y)
  x_scale <- sqrt(
    colMeans(
      sweep(X, 2, x_mean, "-")^2
    )
  )

  X_std <- sweep(
    X, 2, x_mean, "-"
  )

  X_std <- sweep(
    X_std, 2, x_scale, "/"
  )

  y_centered <- y - y_mean

  final_fit <- glmnet(
    X_std,
    y_centered,
    alpha = 1,
    lambda = best_lambda,
    standardize = FALSE,
    intercept = FALSE
  )

  beta <- as.numeric(
    coef(final_fit, s = best_lambda)
  )[-1]

  return(list(
    fit = final_fit,
    lambda = best_lambda,
    beta = beta,
    x_mean = x_mean,
    x_scale = x_scale,
    y_mean = y_mean
  ))
}

X_test <- 500
knn5 <- function(X_train, y_train, X_test, distance = "euclidean") {
  predictions <- numeric(nrow(X_test))
  for (i in 1:nrow(X_test)) {
    if (distance == "euclidean") {
      distances <- sqrt(rowSums((X_train - 
                                   matrix(X_test[i, ], 
                                          nrow = nrow(X_train),
                                          ncol = ncol(X_train),
                                          byrow = TRUE))^2))

    } else if (distance == "manhattan") {
      distances <- rowSums(abs(X_train - 
                                matrix(X_test[i, ], 
                                       nrow = nrow(X_train),
                                       ncol = ncol(X_train),
                                       byrow = TRUE)))

    } else {
      stop("distance must be 'euclidean' or 'manhattan'")
    }
    nearest <- order(distances)[1:5]
    nearest_labels <- y_train[nearest]
    vote_counts <- table(factor(nearest_labels, levels = 0:9))
    predictions[i] <- which.max(vote_counts) - 1
  }
  return(predictions)
}
